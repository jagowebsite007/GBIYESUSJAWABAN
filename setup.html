/** * Inisialisasi Database dengan metode Safe Migrate * Mencegah penghapusan/clear() data eksisting pengguna. */ function setupDatabase() { var ss = SpreadsheetApp.getActiveSpreadsheet(); for (var sheetName in DATABASE_SCHEMA) { var headers = DATABASE_SCHEMA[sheetName]; var sheet = ss.getSheetByName(sheetName); if (!sheet) { // Jika sheet belum ada, buat sheet baru sheet = ss.insertSheet(sheetName); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#2C5F8A').setFontColor('#FFFFFF'); } else { // Safe Migrate: Update header baris pertama tanpa menghapus data var currentHeadersRange = sheet.getRange(1, 1, 1, headers.length); currentHeadersRange.setValues([headers]); currentHeadersRange.setFontWeight('bold').setBackground('#2C5F8A').setFontColor('#FFFFFF'); } // Kirim sampel data dummy jika sheet masih kosong if (sheet.getLastRow() <= 1) { populateDummyData(sheet, sheetName); } sheet.autoResizeColumns(1, headers.length); } SpreadsheetApp.flush(); Logger.log('Inisialisasi Safe Migrate Database GBI Yesus Jawaban berhasil.');
